   174→              className="absolute -left-[19px] top-1.5 h-2 w-2 rounded-full ring-2 ring-background"

   176→              aria-hidden
   177→            />
   178→            <div className="rounded-md border border-border/60 bg-background/40 px-2.5 py-1.5">

   209→/* ── quantities (Phase 5/6/9): dispatched / in-transit / received / outstanding ── */
   210→function QuantityBlock({ ship }: { ship: TraceShip }) {
   211→  const q = ship.quantities;
   212→  const has = q.dispatched.length || q.inTransit.length || q.received.length || q.outstanding.length;
   213→  if (!has) return <EmptyState message="No allocated cargo lines — quantity not derivable from records" />;
   214→  const row = (label: string, rows: { material: string; uom: string; quantity: number }[], tone: string) => (
   215→    <div className="flex items-start justify-between gap-3 border-b border-border/40 py-1.5 last:border-0">
   216→      <span className="shrink-0 text-[11px] text-muted-foreground">{label}</span>
   217→      <span className="min-w-0 text-right text-[11px] font-medium tabular-nums">
   218→        {rows.length === 0
   219→          ? "—"
   220→          : rows.map((r, i) => (
   221→              <span key={`${r.material}-${i}`} className={tone}>
   222→                {r.quantity.toLocaleString()} {r.uom} <span className="font-normal text-muted-foreground">{r.material}</span>
   223→                {i < rows.length - 1 ? " · " : ""}
   224→              </span>
   225→            ))}
   226→      </span>
   227→    </div>
   228→  );
   229→  return (
   230→    <div>
   231→      {row("Dispatched (PO-allocated)", q.dispatched, "")}
   232→      {row("Recorded in transit", q.inTransit, "")}
   233→      {row("Received (GRN)", q.received.map((r) => ({ material: `${r.material}${r.ref ? ` (${r.ref})` : ""}`, uom: r.uom, quantity: r.quantity })), "")}
   234→      {row("Outstanding", q.outstanding, "text-amber-300")}
   235→      <p className="mt-1.5 text-[10px] leading-relaxed text-muted-foreground">
   236→        Quantities stay in their recorded unit of measure — no conversion without a documented basis.
   237→      </p>
   238→    </div>
   239→  );
   240→}
