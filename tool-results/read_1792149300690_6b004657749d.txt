    48→function PriorityChip({ value }: { value: string }) {
    49→  const map: Record<string, string> = {
    50→    CRITICAL: "border-rose-300 bg-rose-100 text-rose-800 dark:border-rose-400/25 dark:bg-rose-400/10 dark:text-rose-300",
    51→    HIGH: "border-orange-300 bg-orange-100 text-orange-800 dark:border-orange-400/25 dark:bg-orange-400/10 dark:text-orange-300",
    52→  };
    53→  return (
    54→    <span
    55→      className={`inline-flex items-center whitespace-nowrap rounded-md border px-1.5 py-0.5 text-[10px] font-semibold ${
    56→        map[value] ?? "border-zinc-300 bg-zinc-100 text-zinc-700 dark:border-zinc-400/25 dark:bg-zinc-900 dark:text-zinc-300"
    57→      }`}
    58→    >
    59→      {value}
    60→    </span>
    61→  );
    62→}
