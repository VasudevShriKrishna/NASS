   124→function ModeIcon({ mode }: { mode: string }) {
   125→  const m = (mode ?? "").toUpperCase();
   126→  if (m === "AIR") return <Plane className="h-3.5 w-3.5 text-zinc-500" aria-label="Air freight" />;

   131→/** ETA estimate type badge — violet for manual (never imply authority), teal for model (Rule 15). */
   132→function EtaTypeBadge({ type }: { type: string }) {
   133→  const t = (type ?? "").toLowerCase();
   134→  const cls =
   135→    t === "manual"
   136→      ? "border-violet-300 bg-violet-100 text-violet-800 dark:border-violet-400/25 dark:bg-violet-400/10 dark:text-violet-300"
   137→      : t === "model"
   138→        ? "border-teal-300 bg-teal-100 text-teal-800 dark:border-teal-400/25 dark:bg-teal-400/10 dark:text-teal-300"
   139→        : ZINC_CHIP;
   140→  return <span className={`${CHIP} ${cls}`}>{t || "unknown"}</span>;
   141→}

   143→/** Identity status colors per spec: RESOLVED emerald / UNRESOLVED rose / MISSING rose. */
   144→function identityPillCls(status: string): string {
   145→  if (status === "RESOLVED")
   146→    return "border-emerald-300 bg-emerald-100 text-emerald-800 dark:border-emerald-400/25 dark:bg-emerald-400/10 dark:text-emerald-300";
   147→  return "border-rose-300 bg-rose-100 text-rose-800 dark:border-rose-400/25 dark:bg-rose-400/10 dark:text-rose-300";
   148→}

   150→/** Vessel position recency — stale when older than 12 h (M11 cadence). */
   151→function vesselFreshness(at: string | null | undefined): string {
   152→  if (!at) return "unavailable";
   153→  const hrs = (Date.now() - new Date(at).getTime()) / 3_600_000;
   154→  if (Number.isNaN(hrs)) return "unavailable";
   155→  if (hrs <= 6) return "fresh";
   156→  if (hrs <= 12) return "aging";
   157→  return "stale";
   158→}
   159→
   160→function slipDays(displayEta: string | null, promised: string | null): number | null {
   161→  if (!displayEta || !promised) return null;
   162→  const ms = new Date(displayEta).getTime() - new Date(promised).getTime();
   163→  if (!(ms > 0)) return null;
   164→  return Math.ceil(ms / 86_400_000);
   165→}

   175→export function ImportTrackingView({ drillRef }: { drillRef?: string | null }) {
   176→  const { data, loading, error } = useApi<ShipmentRow[]>("/api/shipments");
   177→  const drillTo = useAppStore((s) => s.drillTo);
   178→  const openTrace = useAppStore((s) => s.openTrace);
   179→  const shipments = useMemo(() => data ?? [], [data]);
   180→  /** user pick tagged with the drill context it was made in — a NEW drillRef supersedes it */
   181→  const [pick, setPick] = useState<{ id: string; atDrill: string | null } | null>(null);
   182→
   183→  // Fully derived selection (no effects): drill ref > valid user pick > SHP-8802 story > first record.
   184→  const selected = useMemo(() => {
   185→    if (pick && pick.atDrill === (drillRef ?? null)) {
   186→      const picked = shipments.find((s) => s.id === pick.id);
   187→      if (picked) return picked;
   188→    }
   189→    const byDrill = drillRef ? shipments.find((s) => s.ref === drillRef) : undefined;
   190→    return byDrill ?? shipments.find((s) => s.ref === "SHP-8802") ?? shipments[0] ?? null;
   191→  }, [shipments, drillRef, pick]);
   192→
   193→  if (loading) return <LoadingState label="Loading shipments, milestones and ETA provenance…" />;
   194→  if (error) return <ErrorState message={`Failed to load shipments: ${error}`} />;
   195→
   196→  return (
   197→    <div className="grid items-start gap-4 xl:grid-cols-[380px_1fr]">
   198→      {/* ── master list ── */}
   199→      <section aria-label="Shipment list" className="rounded-xl border bg-card">
   200→        <div className="flex items-center justify-between border-b px-4 py-3">
   201→          <h2 className="text-sm font-semibold">Import shipments</h2>
   202→          <div className="flex items-center gap-2">
   203→            <span className="text-xs tabular-nums text-muted-foreground">{shipments.length} records</span>
   204→            <Button
   205→              variant="outline"
   206→              size="sm"
   207→              className="h-7 gap-1 border-primary/30 px-2 text-[11px] text-primary hover:bg-primary/10 hover:text-primary"
   208→              onClick={() => openTrace()}
   209→              title="Open the NASS Freight Traceability Map"
   210→            >
   211→              <RadioTower className="h-3 w-3" /> Live Map
   212→            </Button>
   213→          </div>
   214→        </div>
   215→        <div className={`max-h-[460px] overflow-y-auto p-2 xl:max-h-[calc(100vh-140px)] ${SCROLL}`}>
   216→          {shipments.length === 0 ? (
   217→            <EmptyState message="No shipments found" />
   218→          ) : (
   219→            <ul className="space-y-1.5">
   220→              {shipments.map((s) => {
   221→                const slip = slipDays(s.displayEta, s.supplierPromisedEta);
   222→                const isSel = s.id === selected?.id;
   223→                return (
   224→                  <li key={s.id}>
   225→                    <button
   226→                      type="button"
   227→                      onClick={() => setPick({ id: s.id, atDrill: drillRef ?? null })}
   228→                      aria-current={isSel ? "true" : undefined}
   229→                      className={`w-full rounded-lg border p-3 text-left transition-colors ${
   230→                        isSel
   231→                          ? "border-zinc-500 bg-muted/70 dark:border-zinc-300"
   232→                          : "border-transparent hover:border-zinc-300 hover:bg-muted/40 dark:hover:border-zinc-700"
   233→                      }`}
   234→                    >
   235→                      <div className="flex items-center gap-2">
   236→                        <ModeIcon mode={s.mode} />
   237→                        <span className="font-mono text-sm font-bold">{s.ref}</span>
   238→                        <span className="ml-auto flex items-center gap-1.5">
   239→                          <StatusPill value={s.status} />
   240→                        </span>
   241→                      </div>
   242→                      <p className="mt-1 truncate text-xs text-muted-foreground">
   243→                        {s.originPort ?? "?"} → {s.destinationPort ?? "?"}

   248→                          {s.supplierPromisedEta ? ` · promised ${fmtDate(s.supplierPromisedEta)}` : ""}
   249→                        </span>
   250→                        {slip != null && (

   252→                            +{slip}d slip

   255→                        <FreshnessBadge value={s.freshness} />
   256→                        {s.exceptionCases.length > 0 && (
   257→                          <span className={`${CHIP} border-rose-300 bg-rose-50 text-rose-700 dark:border-rose-400/25 dark:bg-rose-400/10/60 dark:text-rose-300`}>

   266→              })}
   267→            </ul>

   271→
   272→      {/* ── detail ── */}
   273→      <section aria-label="Shipment detail" className="space-y-4">

   278→        )}
   279→      </section>
   280→    </div>
   281→  );
