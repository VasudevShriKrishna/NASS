   188→export function OperationMap({
   189→  ships,
   190→  simBase,
   191→  factor,
   192→  selectedRef,
   193→  onPick,
   194→  onBackgroundClick,
   195→  onSitePick,
   196→}: {
   197→  ships: TraceShip[];
   198→  simBase: { server: number; wall: number } | null;
   199→  factor: number;
   200→  selectedRef: string | null;
   201→  onPick: (ref: string) => void;
   202→  onBackgroundClick?: () => void;
   203→  onSitePick: (pick: SitePick) => void;
   204→}) {
   205→  const containerRef = useRef<HTMLDivElement>(null);
   206→  const labelLayerRef = useRef<HTMLDivElement>(null);
   207→  const tooltipRef = useRef<HTMLDivElement>(null);
   208→  const [noWebGL, setNoWebGL] = useState(false);
   209→
   210→  const mapRef = useRef<MlMap | null>(null);
   211→  const readyRef = useRef(false);
   212→  const disposedRef = useRef(false);
   213→  const rafRef = useRef(0);
   214→  const timersRef = useRef<number[]>([]);
   215→
   216→  const shipsRef = useRef<TraceShip[]>(ships);
   217→  const simBaseRef = useRef(simBase);
   218→  const factorRef = useRef(factor);
   219→  const selectedRefRef = useRef<string | null>(selectedRef);
   220→  const normRef = useRef<Map<string, number>>(new Map());
   221→  const reducedRef = useRef(false);
   222→  const cbRef = useRef({ onPick, onBackgroundClick, onSitePick });
   223→  const dashPhaseRef = useRef(0);
   224→
   225→  const vesselsRef = useRef<Map<string, VesselEntry>>(new Map());
   226→  const arcsRef = useRef<Map<string, MaritimeRoute>>(new Map());
   227→  const portsRef = useRef<Map<string, DotEntry>>(new Map());
   228→  const sitesRef = useRef<Map<string, { kind: "supplier" | "warehouse"; code: string; entry: DotEntry }>>(new Map());
   229→  const labelsRef = useRef<Map<string, LabelEntry>>(new Map());
   230→  const tipRef = useRef<{ ref: string | null; lastBuild: number }>({ ref: null, lastBuild: 0 });
   231→
   232→  /* mirror latest props into refs (after commit, before any effect work) */
   233→  useEffect(() => {
   234→    shipsRef.current = ships;
   235→    simBaseRef.current = simBase;
   236→    factorRef.current = factor;
   237→    selectedRefRef.current = selectedRef;
   238→    cbRef.current = { onPick, onBackgroundClick, onSitePick };
   239→    normRef.current = valueNorms(ships);
   240→  });
   241→  const computeSimNow = () => {
   242→    const b = simBaseRef.current;
   243→    return b ? b.server + (Date.now() - b.wall) * factorRef.current : Date.now();
   244→  };
   245→
   246→  /* ── route + feeder GeoJSON refresh ────────────────────────────────── */
   247→  const refreshRoutes = () => {
   248→    const map = mapRef.current;
   249→    if (!map || !readyRef.current) return;
   250→    const simNow = computeSimNow();
   251→    const sel = selectedRefRef.current;
   252→
   253→    const traveled: LineFeature[] = [];
   254→    const remaining: LineFeature[] = [];
   255→    for (const [ref, v] of vesselsRef.current) {
   256→      const p = reducedRef.current ? v.ship.progress : liveProgressAt(v.ship, simNow);
   257→      v.p = p;
   258→      const c = STATUS_COLOR[v.ship.liveStatus];
   259→      const w = (normRef.current.get(ref) ?? 0.6) * 1.5 + 0.75;
   260→      const o = !sel ? 0.9 : ref === sel ? 1 : 0.12;
   261→      const or = !sel ? 0.3 : ref === sel ? 0.45 : 0.05;
   262→      const of = !sel ? 0.5 : ref === sel ? 0.8 : 0.06;
   263→      if (p > 0.004) traveled.push(line(v.arc.traveledCoords(p), { ref, color: c, o, w, of }));
   264→      if (p < 0.996) remaining.push(line(v.arc.remainingCoords(p), { ref, color: c, w: w * 0.9, or }));
   265→    }
   266→    const tSrc = map.getSource("src-traveled") as GeoJSONSource | undefined;
   267→    const rSrc = map.getSource("src-remaining") as GeoJSONSource | undefined;
   268→    tSrc?.setData({ type: "FeatureCollection", features: traveled });
   269→    rSrc?.setData({ type: "FeatureCollection", features: remaining });
   270→
   271→    const feeders: LineFeature[] = [];
   272→    for (const ship of shipsRef.current) {
   273→      const oFor = (mine: boolean) => (!sel ? 0.5 : mine ? 0.9 : 0.06);
   274→      const sSite = ship.supplier ? SUPPLIER_SITES[ship.supplier.code] : null;
   275→      if (sSite && ship.from) {
   276→        feeders.push(
   277→          line(
   278→            [
   279→              [sSite.lng, sSite.lat],
   280→              [ship.from.lng, ship.from.lat],
   281→            ],
   282→            { color: "#d4af37", o: oFor(ship.ref === sel) },
   283→          ),
   284→        );
   285→      }
   286→      const wSite = ship.destinationWarehouse ? WAREHOUSE_SITES[ship.destinationWarehouse.code] : null;
   287→      if (wSite && ship.to) {
   288→        feeders.push(
   289→          line(
   290→            [
   291→              [ship.to.lng, ship.to.lat],
   292→              [wSite.lng, wSite.lat],
   293→            ],
   294→            { color: "#34d399", o: oFor(ship.ref === sel) },
   295→          ),
   296→        );
   297→      }
   298→    }
   299→    const fSrc = map.getSource("src-feeders") as GeoJSONSource | undefined;
   300→    fSrc?.setData({ type: "FeatureCollection", features: feeders });
   301→  };
   302→
   303→  /* ── labels: greedy collision engine ───────────────────────────────── */
   304→  const updateLabels = () => {
   305→    const map = mapRef.current;
   306→    const layer = labelLayerRef.current;
   307→    if (!map || !layer) return;
   308→    const zoom = map.getZoom();
   309→    const w = layer.clientWidth || map.getContainer().clientWidth;
   310→    const h = layer.clientHeight || map.getContainer().clientHeight;
   311→    const sel = selectedRefRef.current;
   312→
   313→    const rects: { x: number; y: number; w: number; h: number; id: string }[] = [];
   314→    const push = (cx: number, cy: number, rw: number, rh: number, id: string) =>
   315→      rects.push({ x: cx - rw / 2, y: cy - rh / 2, w: rw, h: rh, id });
   316→    for (const v of vesselsRef.current.values()) push(v.px, v.py, 26, 26, `ship:${v.ship.ref}`);
   317→    for (const [key, p] of portsRef.current) push(p.px, p.py, 13, 13, `port:${key}`);
   318→    for (const [id, s] of sitesRef.current) push(s.entry.px, s.entry.py, 24, 24, id);
   319→
   320→    const items = [...labelsRef.current.values()].sort((a, b) => a.priority - b.priority);
   321→    for (const it of items) {
   322→      let minZoom = it.baseMinZoom;
   323→      if (sel && it.id !== `ship:${sel}`) minZoom += 0.9;
   324→      const [lng, lat] = it.anchor();
   325→      const proj = map.project([lng, lat]);
   326→      /* clustered vessels: anchor to the spiderfy-offset screen position */
   327→      let pt: { x: number; y: number } = { x: proj.x, y: proj.y };
   328→      if (it.kind === "vessel") {
   329→        const v = vesselsRef.current.get(it.id.slice(5));
   330→        if (v && (v.px || v.py)) pt = { x: v.px, y: v.py };
   331→      }
   332→      const off = pt.x < 34 || pt.y < 26 || pt.x > w - 34 || pt.y > h - 58;
   333→      if (!Number.isFinite(pt.x) || off || zoom < minZoom) {
   334→        it.el.classList.remove(styles.labelVisible);
   335→        it.leaderEl.style.opacity = "0";
   336→        continue;
   337→      }
   338→      const charW = it.kind === "vessel" ? 6.5 : 5.7;
   339→      const bw = it.text.length * charW + 10;
   340→      const bh = 14;
   341→      const cands = [
   342→        { x: pt.x - bw / 2, y: pt.y - bh - 12, side: 0 },
   343→        { x: pt.x + 13, y: pt.y - bh / 2, side: 1 },
   344→        { x: pt.x - bw / 2, y: pt.y + 11, side: 2 },
   345→        { x: pt.x - bw - 13, y: pt.y - bh / 2, side: 3 },
   346→      ];
   347→      let placed: { x: number; y: number; box: { x: number; y: number; w: number; h: number } } | null = null;
   348→      for (const c of cands) {
   349→        const box = { x: c.x - 3, y: c.y - 2, w: bw + 6, h: bh + 4 };
   350→        if (!rects.some((r) => r.id !== it.id && r.x < box.x + box.w && r.x + r.w > box.x && r.y < box.y + box.h && r.y + r.h > box.y)) {
   351→          placed = { x: c.x, y: c.y, box };
   352→          break;
   353→        }
   354→      }
   355→      if (!placed) {
   356→        it.el.classList.remove(styles.labelVisible);
   357→        it.leaderEl.style.opacity = "0";
   358→        continue;
   359→      }
   360→      rects.push({ ...placed.box, id: it.id });
   361→      it.el.style.transform = `translate3d(${placed.x.toFixed(1)}px, ${placed.y.toFixed(1)}px, 0)`;
   362→      it.el.classList.add(styles.labelVisible);
   363→      /* leader line from anchor toward the label when displaced */
   364→      const ex = placed.x + bw / 2;
   365→      const ey = placed.y + bh / 2;
   366→      const dx = ex - pt.x;
   367→      const dy = ey - pt.y;
   368→      const dist = Math.hypot(dx, dy);
   369→      if (dist > 15) {
   370→        it.leaderEl.style.transform = `translate3d(${pt.x.toFixed(1)}px, ${pt.y.toFixed(1)}px, 0) rotate(${((Math.atan2(dy, dx) * 180) / Math.PI).toFixed(1)}deg)`;
   371→        it.leaderEl.style.width = `${Math.min(dist - 6, 30).toFixed(0)}px`;
   372→        it.leaderEl.style.opacity = "0.8";
   373→      } else {
   374→        it.leaderEl.style.opacity = "0";
   375→      }
   376→    }
   377→  };
   378→
   379→  /* ── vessels: glide + spiderfy + cluster chips ─────────────────────── */
   380→  const updateVessels = (simNow: number) => {
   381→    const map = mapRef.current;
   382→    if (!map) return;
   383→    const entries = [...vesselsRef.current.values()];
   384→    const pts: { x: number; y: number }[] = [];
   385→    for (const v of entries) {
   386→      v.p = reducedRef.current ? v.ship.progress : liveProgressAt(v.ship, simNow);
   387→      v.pos = v.arc.pointAt(v.p);
   388→      v.marker.setLngLat([v.pos.lng, v.pos.lat]);
   389→      const pt = map.project([v.pos.lng, v.pos.lat]);
   390→      pts.push(pt);
   391→      v.rotEl.style.transform = `rotate(${v.pos.angle.toFixed(1)}deg)`;
   392→    }
   393→    /* spiderfy overlapping vessels (same corridor at low zoom) */
   394→    const parent = entries.map((_, i) => i);
   395→    const find = (i: number): number => (parent[i] === i ? i : (parent[i] = find(parent[i])));
   396→    for (let i = 0; i < entries.length; i++) {
   397→      for (let j = i + 1; j < entries.length; j++) {
   398→        if (Math.hypot(pts[i].x - pts[j].x, pts[i].y - pts[j].y) < 26) parent[find(i)] = find(j);
   399→      }
   400→    }
   401→    const groups = new Map<number, number[]>();
   402→    entries.forEach((_, i) => {
   403→      const root = find(i);
   404→      const g = groups.get(root);
   405→      if (g) g.push(i);
   406→      else groups.set(root, [i]);
   407→    });
   408→    for (const g of groups.values()) {
   409→      const n = g.length;
   410→      if (n < 2) {
   411→        const v = entries[g[0]];
   412→        v.marker.setOffset([0, 0]);
   413→        v.clusterSize = 1;
   414→        v.px = pts[g[0]].x;
   415→        v.py = pts[g[0]].y;
   416→        v.el.classList.remove(styles.clustered);
   417→        continue;
   418→      }
   419→      const cx = g.reduce((s, i) => s + pts[i].x, 0) / n;
   420→      const cy = g.reduce((s, i) => s + pts[i].y, 0) / n;
   421→      const radius = 15 + n * 3.2;
   422→      const ordered = [...g].sort(
   423→        (a, b) => Math.atan2(pts[a].y - cy, pts[a].x - cx) - Math.atan2(pts[b].y - cy, pts[b].x - cx),
   424→      );
   425→      ordered.forEach((idx, k) => {
   426→        const a = (2 * Math.PI * k) / n - Math.PI / 2 + Math.atan2(pts[ordered[0]].y - cy, pts[ordered[0]].x - cx);
   427→        const tx = cx + radius * Math.cos(a);
   428→        const ty = cy + radius * Math.sin(a);
   429→        const v = entries[idx];
   430→        v.marker.setOffset([tx - pts[idx].x, ty - pts[idx].y]);
   431→        v.px = tx;
   432→        v.py = ty;
   433→        v.clusterSize = n;
   434→        v.el.classList.add(styles.clustered);
