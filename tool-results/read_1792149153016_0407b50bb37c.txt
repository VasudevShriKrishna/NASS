    30→const CASE_STATES_LINEAR = [
    31→  "DETECTED",

    42→
    43→interface TransitionOption {
    44→  to: string;
    45→  reopen?: boolean;
    46→}
    47→function allowedTransitions(state: string): TransitionOption[] {
    48→  const out: TransitionOption[] = [];
    49→  if (state === "RESOLVED") {
    50→    out.push({ to: "INVESTIGATING", reopen: true });
    51→  } else if (state !== "CANCELLED") {
    52→    const idx = CASE_STATES_LINEAR.indexOf(state as (typeof CASE_STATES_LINEAR)[number]);
    53→    if (idx >= 0 && idx < CASE_STATES_LINEAR.length - 1) out.push({ to: CASE_STATES_LINEAR[idx + 1] });
    54→    out.push({ to: "CANCELLED" });
    55→  }
    56→  return out;
    57→}

   200→/* ── Transition control (interactive core) ── */
   201→function TransitionPanel({ kase, events, onSuccess }: { kase: CaseDetail; events: CaseEventRow[]; onSuccess: () => void }) {
   202→  const { toast } = useToast();
   203→  const options = useMemo(() => allowedTransitions(kase.state), [kase.state]);
   204→  const [toState, setToState] = useState<string>(options[0]?.to ?? "");
   205→  const [actor, setActor] = useState("Ops Supervisor");
   206→  const [note, setNote] = useState("");
   207→  const [busy, setBusy] = useState(false);
   208→
   209→  const isReopen = kase.state === "RESOLVED" && toState === "INVESTIGATING";
   210→  const noteRequired = toState === "RESOLVED" || isReopen;
   211→  const visited = useMemo(() => {
   212→    const set = new Set<string>([kase.state]);
   213→    for (const e of events) {
   214→      if (e.fromState) set.add(e.fromState);
   215→      if (e.toState) set.add(e.toState);
   216→    }
   217→    return Array.from(set);
   218→  }, [events, kase.state]);
   219→
   220→  async function submit() {
   221→    if (!toState || busy) return;
   222→    if (!actor.trim()) {
   223→      toast({ title: "Actor required", description: "The audit trail (M22) must record who performed this transition.", variant: "destructive" });
   224→      return;
   225→    }
   226→    if (noteRequired && !note.trim()) {
   227→      toast({
   228→        title: "Closure evidence / reason required",
   229→        description: isReopen ? "Reopening a resolved case requires the regression reason." : "A case can only reach RESOLVED with verified closure evidence (M17).",
   230→        variant: "destructive",
   231→      });
   232→      return;
   233→    }
   234→    setBusy(true);
   235→    const res = await postJson<{ case: CaseRow; transition: { from: string; to: string } }>(`/api/exceptions/${kase.id}/transition`, {
   236→      toState,
   237→      actor: actor.trim(),
   238→      note: note.trim() || undefined,
   239→    });
   240→    setBusy(false);
   241→    if (res.ok) {
   242→      toast({
   243→        title: `Case ${kase.ref} advanced`,
   244→        description: `${res.data?.transition.from ?? kase.state} → ${toState}${isReopen ? " (REOPEN)" : ""} recorded by ${actor.trim()}.`,
   245→      });
   246→      setNote("");
   247→      setToState("");
   248→      onSuccess();
   249→    } else {
   250→      toast({ title: "Transition rejected", description: res.error?.message ?? "Invalid transition.", variant: "destructive" });
   251→    }
   252→  }
   253→
   254→  return (
   255→    <SectionCard
   256→      title={
   257→        <span className="inline-flex items-center gap-1.5">
   258→          <ShieldCheck className="h-4 w-4 text-teal-700 dark:text-teal-400" /> Advance closed loop (M17)
   259→        </span>
   260→      }
   261→      subtitle="Linear order with CANCELLED exit; RESOLVED may reopen to INVESTIGATING on regression. Server validates every transition."
   262→    >
   263→      <div className="mb-4">
   264→        <p className="mb-1.5 text-xs font-medium text-muted-foreground">Case journey</p>
   265→        <MiniFlow state={kase.state} visited={visited} />
   266→      </div>
   267→      <div className="grid gap-3 sm:grid-cols-2">
   268→        <div className="space-y-1.5">
   269→          <label className="text-xs font-medium text-muted-foreground" htmlFor="next-state">
   270→            Next state (allowed from {kase.state})
   271→          </label>
   272→          <Select value={toState} onValueChange={setToState}>
   273→            <SelectTrigger id="next-state" aria-label="Next state">
   274→              <SelectValue placeholder="Select next state" />
   275→            </SelectTrigger>
   276→            <SelectContent>
   277→              {options.map((o) => (
   278→                <SelectItem key={o.to} value={o.to}>
   279→                  {o.reopen ? "INVESTIGATING — REOPEN: condition regressed" : o.to.replaceAll("_", " ")}
   280→                </SelectItem>
   281→              ))}
   282→            </SelectContent>
   283→          </Select>
   284→        </div>
   285→        <div className="space-y-1.5">
   286→          <label className="text-xs font-medium text-muted-foreground" htmlFor="actor">
   287→            Actor (recorded in audit trail)
   288→          </label>
   289→          <Input id="actor" value={actor} onChange={(e) => setActor(e.target.value)} placeholder="Ops Supervisor" />
   290→        </div>
   291→        <div className="space-y-1.5 sm:col-span-2">
   292→          <label className="text-xs font-medium text-muted-foreground" htmlFor="transition-note">
   293→            Note {noteRequired ? <span className="font-semibold text-rose-600 dark:text-rose-400">— closure evidence / reason (required)</span> : "(optional)"}
   294→          </label>
   295→          <Textarea
   296→            id="transition-note"
   297→            value={note}
   298→            onChange={(e) => setNote(e.target.value)}
   299→            placeholder={noteRequired ? "e.g. GRN-5003 read-back confirms full receipt; ledger reconciled" : "Context for the case timeline"}
   300→            rows={3}
   301→          />
   302→        </div>
   303→      </div>
   304→      <div className="mt-3 flex items-center justify-between gap-3">
   305→        <p className="text-xs text-muted-foreground">
   306→          Every transition writes a CaseEvent and an append-only AuditEvent with a correlation id.
   307→        </p>
   308→        <Button type="button" onClick={submit} disabled={busy || !toState} className="min-h-9 shrink-0">
   309→          {busy ? <span className="inline-block h-4 w-4 animate-spin rounded-full border-2 border-current border-t-transparent" aria-hidden /> : <Send className="h-4 w-4" aria-hidden />}
   310→          Submit transition
   311→        </Button>
   312→      </div>
   313→    </SectionCard>
   314→  );
   315→}
