   112→/* ── Approval gateway (interactive) ── */
   113→function DecisionPanel({ action, onDone }: { action: ActionRow; onDone: () => void }) {
   114→  const { toast } = useToast();
   115→  const [approver, setApprover] = useState("Procurement Lead");
   116→  const [role, setRole] = useState("Approver");
   117→  const [comment, setComment] = useState("");
   118→  const [busy, setBusy] = useState<"APPROVED" | "REJECTED" | null>(null);
   119→
   120→  async function decide(decision: "APPROVED" | "REJECTED") {
   121→    if (busy) return;
   122→    if (!approver.trim()) {
   123→      toast({ title: "Approver required", description: "The approval record must identify the accountable approver (M18).", variant: "destructive" });
   124→      return;
   125→    }
   126→    setBusy(decision);
   127→    const res = await postJson<{ action: ActionRow; decision: string; correlation_id: string }>(`/api/actions/${action.id}/decide`, {
   128→      decision,
   129→      approver: approver.trim(),
   130→      role: role.trim() || "Approver",
   131→      comment: comment.trim() || undefined,
   132→    });
   133→    setBusy(null);
   134→    if (res.ok) {
   135→      toast({
   136→        title: decision === "APPROVED" ? `${action.ref} approved and submitted` : `${action.ref} rejected`,
   137→        description:
   138→          decision === "APPROVED"
   139→            ? "Gateway submitted the command with its idempotency key; outcome is PENDING_VERIFICATION until target read-back."
   140→            : "Command cancelled with reason; the linked case returns to ACTION_PROPOSED for rework.",
   141→      });
   142→      setComment("");
   143→      onDone();
   144→    } else {
   145→      toast({ title: "Gateway decision failed", description: res.error?.message ?? "Unknown gateway error.", variant: "destructive" });
   146→    }
   147→  }
   148→
   149→  return (
   150→    <div className="space-y-3 rounded-lg border border-teal-300 bg-teal-50/60 p-3 dark:border-teal-400/25 dark:bg-teal-400/10">
   151→      <p className="text-xs font-semibold uppercase tracking-wide text-teal-800 dark:text-teal-300">Approval gateway — decide this action</p>
   152→      <div className="grid gap-3 sm:grid-cols-2">
   153→        <div className="space-y-1.5">
   154→          <label className="text-xs font-medium text-muted-foreground" htmlFor="approver">
   155→            Approver
   156→          </label>
   157→          <Input id="approver" value={approver} onChange={(e) => setApprover(e.target.value)} placeholder="Procurement Lead" />
   158→        </div>
   159→        <div className="space-y-1.5">
   160→          <label className="text-xs font-medium text-muted-foreground" htmlFor="approver-role">
   161→            Role
   162→          </label>
   163→          <Input id="approver-role" value={role} onChange={(e) => setRole(e.target.value)} placeholder="Approver" />
   164→        </div>
   165→        <div className="space-y-1.5 sm:col-span-2">
   166→          <label className="text-xs font-medium text-muted-foreground" htmlFor="approval-comment">
   167→            Comment (recorded on the approval)
   168→          </label>
   169→          <Textarea id="approval-comment" value={comment} onChange={(e) => setComment(e.target.value)} rows={2} placeholder="Rationale, conditions or rejection reason" />
   170→        </div>
   171→      </div>
   172→      <div className="flex flex-wrap items-center gap-2">
   173→        <Button type="button" disabled={busy !== null} onClick={() => decide("APPROVED")} className="bg-emerald-600 text-white hover:bg-emerald-700">
   174→          {busy === "APPROVED" ? <span className="inline-block h-4 w-4 animate-spin rounded-full border-2 border-current border-t-transparent" aria-hidden /> : <Check className="h-4 w-4" aria-hidden />}
   175→          Approve
   176→        </Button>
   177→        <Button type="button" disabled={busy !== null} onClick={() => decide("REJECTED")} variant="outline" className="border-rose-400 text-rose-700 hover:bg-rose-50 hover:text-rose-800 dark:border-rose-400/25 dark:text-rose-400 dark:hover:bg-rose-400/10">
   178→          {busy === "REJECTED" ? <span className="inline-block h-4 w-4 animate-spin rounded-full border-2 border-current border-t-transparent" aria-hidden /> : <X className="h-4 w-4" aria-hidden />}
   179→          Reject
   180→        </Button>
   181→      </div>
   182→      <p className="rounded-md border border-amber-300 bg-amber-50 px-2.5 py-2 text-xs text-amber-900 dark:border-amber-400/25 dark:bg-amber-400/10 dark:text-amber-300">
   183→        LLM output is a proposal, not an action authorization — only the action gateway executes after policy, authorization and approval checks (M18).
   184→      </p>
   185→    </div>
   186→  );
   187→}
