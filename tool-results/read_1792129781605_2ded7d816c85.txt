   457→  const setView = useAppStore((s) => s.setView);
   458→  const clearDrill = useAppStore((s) => s.clearDrill);
   459→  const traceShip = useAppStore((s) => s.traceShip);
   460→  const clearTraceShip = useAppStore((s) => s.clearTraceShip);
   461→  const { data: snap, loading, error, refresh, syncedAt } = useApi<TraceSnapshot>("/api/live-trace", [], 0);
   462→  const [picked, setPicked] = useState<string | null>(null);
   463→  const [factor, setFactor] = useState<number>(600);
   464→  const now = useNow(1000);
   465→
   466→  const activeRef = picked ?? traceShip ?? drillRef ?? null;
   467→
   468→  /* sim clock: server snapshot + real elapsed × time-lapse factor */
   469→  const baseRef = useRef<{ server: number; wall: number } | null>(null);
   470→  useEffect(() => {
   471→    if (snap) baseRef.current = { server: +new Date(snap.serverTime), wall: Date.now() };
   472→  }, [snap]);
   473→  const [simNow, setSimNow] = useState<number | null>(null);
   474→  useEffect(() => {
   475→    const id = setInterval(() => {
   476→      const b = baseRef.current;
   477→      if (b) setSimNow(b.server + (Date.now() - b.wall) * factor);
   478→    }, 150);
   479→    return () => clearInterval(id);
   480→  }, [factor]);
   481→
   482→  /* hourly cadence: refetch exactly at the next top-of-hour + pulse */
   483→  const [flash, setFlash] = useState(0);
   484→  useEffect(() => {
   485→    if (!snap) return;
   486→    const delay = Math.max(1500, +new Date(snap.nextUpdateAt) - Date.now());
   487→    const id = setTimeout(() => {
   488→      refresh();
   489→      setFlash((f) => f + 1);
   490→    }, delay);
   491→    return () => clearTimeout(id);
   492→  }, [snap, refresh]);

   499→    for (const s of snap.ships) {
   500→      if (s.liveStatus === "ARRIVED") {
   501→        m.set(s.ref, 1);
   502→        continue;
   503→      }
   504→      if (s.liveStatus === "BOOKED") {
   505→        m.set(s.ref, 0);
   506→        continue;
   507→      }
   508→      const end = s.actualArrival ? +new Date(s.actualArrival) : s.eta ? +new Date(s.eta) : null;
   509→      const start = s.departure ? +new Date(s.departure) : 0;
   510→      m.set(s.ref, !end ? 0 : clamp01((simNow - start) / Math.max(1, end - start)));
   511→    }
   512→    return m;

   514→
   515→  const byRef = useMemo(() => new Map((snap?.ships ?? []).map((s) => [s.ref, s])), [snap]);
